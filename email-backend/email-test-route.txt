/* POST /api/email/test */

app.post("/api/email/test", async (req, res) => {
  try {
    const transporter = getConfiguredTransporter();

    const info = await transporter.sendMail({
      from: {
        name: "AdNex Media HR",
        address: SMTP_USER,
      },
      to: req.admin.email,
      subject: "AdNex Media SMTP Test",
      text: "Your AdNex Media email backend is able to send email.",
    });

    console.info("SMTP test email sent successfully:", {
      recipient: req.admin.email,
      messageId: info.messageId,
    });

    return res.json({
      success: true,
      message: "SMTP test email sent successfully.",
      messageId: info.messageId,
      recipient: req.admin.email,
    });
  } catch (error) {
    const code = String(error?.code || "");

    console.error("SMTP test error:", {
      message: String(error?.message || "Unknown error"),
      code,
      command: error?.command,
      responseCode: error?.responseCode,
    });

    if (String(error?.message || "").includes("SMTP_PASS is missing")) {
      return res.status(500).json({
        success: false,
        code: "SMTP_CONFIGURATION_ERROR",
        message: "SMTP_PASS is not configured in the backend environment.",
      });
    }

    
if (code === "EAUTH") {
  return res.status(502).json({
    success: false,
    code: "SMTP_AUTH_FAILED",
    message:
      "Gmail rejected SMTP authentication. Check SMTP_USER and your Gmail App Password.",
  });
}

if (
  ["ECONNECTION", "ETIMEDOUT", "ESOCKET", "ECONNREFUSED"].includes(code)
) {
  return res.status(502).json({
    success: false,
    code: "SMTP_CONNECTION_FAILED",
    message:
      "Could not connect to Gmail SMTP. Check smtp.gmail.com, port 465, and TLS settings.",
  });
}


    return res.status(502).json({
      success: false,
      code: "SMTP_TEST_FAILED",
      message: "SMTP test failed. Check the backend logs for details.",
    });
  }
});
